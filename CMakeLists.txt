cmake_minimum_required(VERSION 3.24)
project(Youngkly VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SI_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../SI_generator_관련/SI_Generator_Project/native" CACHE PATH "Existing SI Generator native source")
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Network Xml Gui)
find_package(ZLIB REQUIRED)
configure_file(src/SIParser.cpp.in "${CMAKE_CURRENT_BINARY_DIR}/SIParser.cpp" @ONLY)
configure_file(src/SIModel.cpp.in "${CMAKE_CURRENT_BINARY_DIR}/SIModel.cpp" @ONLY)
add_library(youngkly_core src/Archive.cpp src/Sheet.cpp src/Pipeline.cpp src/FileTransfer.cpp src/GaussianStore.cpp src/Alignment.cpp src/AtomMapping.cpp src/Narrative.cpp src/FutureWork.cpp src/Docx.cpp src/Workspace.cpp src/RelativeTable.cpp src/CaptureCrop.cpp
  "${CMAKE_CURRENT_BINARY_DIR}/SIParser.cpp" "${CMAKE_CURRENT_BINARY_DIR}/SIModel.cpp")
target_include_directories(youngkly_core PUBLIC src "${SI_SOURCE_DIR}/include")
target_link_libraries(youngkly_core PUBLIC Qt6::Core Qt6::Network Qt6::Xml Qt6::Gui ZLIB::ZLIB)
file(SHA256 "${SI_SOURCE_DIR}/src/GaussianParser.cpp" si_parser_hash)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${SI_SOURCE_DIR}/src/GaussianParser.cpp" "${SI_SOURCE_DIR}/include/si/GaussianParser.hpp")
target_compile_definitions(youngkly_core PRIVATE YOUNGKLY_PARSER_VERSION="${si_parser_hash}")
add_executable(Youngkly src/main.cpp)
target_link_libraries(Youngkly PRIVATE youngkly_core)
if(MSVC)
  target_compile_options(youngkly_core PRIVATE /utf-8 /W4)
  target_compile_options(Youngkly PRIVATE /utf-8 /W4)
else()
  target_compile_options(youngkly_core PRIVATE -Wall -Wextra)
endif()
option(YOUNGKLY_BUILD_RENDERER "Build Youngcraft-based batch renderer" ON)
if(YOUNGKLY_BUILD_RENDERER)
  find_package(Qt6 REQUIRED COMPONENTS Gui Widgets OpenGL OpenGLWidgets)
  set(YOUNGCRAFT_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../Youngcraft" CACHE PATH "Existing Youngcraft sources")
  # Generate a local header adapter so camera APIs do not modify Youngcraft itself.
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${YOUNGCRAFT_SOURCE_DIR}/src/render/MoleculeView.h" "${CMAKE_CURRENT_SOURCE_DIR}/src/CameraApi.inc")
  file(READ "${YOUNGCRAFT_SOURCE_DIR}/src/render/MoleculeView.h" yc_header)
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/CameraApi.inc" camera_api)
  string(REPLACE "#include <QQuaternion>" "#include <QQuaternion>\n#include <QJsonObject>\n#include <QJsonArray>\n#include <cmath>" yc_header "${yc_header}")
  string(REPLACE "    void bestView();" "    void bestView();\n${camera_api}" yc_header "${yc_header}")
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/adapter/render")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/adapter/MoleculeView.pending.h" "${yc_header}")
  configure_file("${CMAKE_CURRENT_BINARY_DIR}/adapter/MoleculeView.pending.h" "${CMAKE_CURRENT_BINARY_DIR}/adapter/render/MoleculeView.h" COPYONLY)
  set(yc_sources
    model/Molecule.cpp model/StructureTools.cpp model/GeometryEditor.cpp model/MoleculeEditController.cpp model/FragmentLibrary.cpp
    io/GaussianParser.cpp io/CubeParser.cpp io/NboParser.cpp
    settings/StyleManager.cpp animation/AnimationController.cpp export/ImageExporter.cpp render/IsosurfaceBuilder.cpp
    render/MoleculeView.cpp
    render/moleculeview/MoleculeView_State.cpp render/moleculeview/MoleculeView_Surface.cpp
    render/moleculeview/MoleculeView_Shadows.cpp render/moleculeview/MoleculeView_Scene.cpp
    render/moleculeview/MoleculeView_Overlays.cpp render/moleculeview/MoleculeView_Interaction.cpp
    render/moleculeview/MoleculeView_GL.cpp render/moleculeview/MoleculeView_Camera.cpp
    render/moleculeview/MoleculeView_AtomsBonds.cpp)
  set(yc_wrappers)
  foreach(source IN LISTS yc_sources)
    get_filename_component(stem "${source}" NAME_WE)
    set(YC_INCLUDE "${YOUNGCRAFT_SOURCE_DIR}/src/${source}")
    if(source STREQUAL "export/ImageExporter.cpp")
      set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${YC_INCLUDE}")
      # GIF decoders add dictionary entries one code behind the encoder.
      # Keep this compatibility fix local to Youngkly's adapter.
      file(READ "${YC_INCLUDE}" gif_export_source)
      string(FIND "${gif_export_source}" "if (nextCode == (1 << codeSize) && codeSize < 12)" gif_transition)
      if(gif_transition LESS 0)
        message(FATAL_ERROR "Youngcraft GIF source changed; revalidate the local compatibility adapter")
      endif()
      string(REPLACE "if (nextCode == (1 << codeSize) && codeSize < 12)" "if (nextCode > (1 << codeSize) && codeSize < 12)" gif_export_source "${gif_export_source}")
      file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/yc/${stem}.cpp" "${gif_export_source}")
    else()
      configure_file(src/YoungcraftSource.cpp.in "${CMAKE_CURRENT_BINARY_DIR}/yc/${stem}.cpp" @ONLY)
    endif()
    list(APPEND yc_wrappers "${CMAKE_CURRENT_BINARY_DIR}/yc/${stem}.cpp")
  endforeach()
  foreach(header IN ITEMS render/MoleculeView animation/AnimationController)
    set(moc_header "${YOUNGCRAFT_SOURCE_DIR}/src/${header}.h")
    if(header STREQUAL "render/MoleculeView")
      set(moc_header "${CMAKE_CURRENT_BINARY_DIR}/adapter/render/MoleculeView.h")
    endif()
    get_filename_component(stem "${header}" NAME)
    add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/yc/moc_${stem}.cpp"
      COMMAND Qt6::moc "-I${YOUNGCRAFT_SOURCE_DIR}/src" -o "yc/moc_${stem}.cpp" "${moc_header}"
      DEPENDS "${moc_header}"
      WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
    list(APPEND yc_wrappers "${CMAKE_CURRENT_BINARY_DIR}/yc/moc_${stem}.cpp")
  endforeach()
  add_library(youngkly_youngcraft STATIC ${yc_wrappers})
  target_include_directories(youngkly_youngcraft PUBLIC "${CMAKE_CURRENT_BINARY_DIR}/adapter" "${YOUNGCRAFT_SOURCE_DIR}/src")
  target_link_libraries(youngkly_youngcraft PUBLIC Qt6::Gui Qt6::Widgets Qt6::OpenGL Qt6::OpenGLWidgets)
  add_executable(youngkly_render src/render_main.cpp src/RenderDialog.cpp)
  target_link_libraries(youngkly_render PRIVATE youngkly_core youngkly_youngcraft)
  configure_file(resources/youngkly.ico "${CMAKE_CURRENT_BINARY_DIR}/youngkly.ico" COPYONLY)
  configure_file(resources/youngkly-icon.rc "${CMAKE_CURRENT_BINARY_DIR}/youngkly-icon.rc" COPYONLY)
  set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/youngkly-icon.rc" PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/youngkly.ico")
  if(MINGW)
    # windres' preprocessor cannot handle Unicode absolute paths in inherited -I
    # arguments. The icon resource has no includes; compile it by relative name.
    add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/youngkly-icon.o"
      COMMAND "${CMAKE_RC_COMPILER}" -i youngkly-icon.rc -o youngkly-icon.o
      DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/youngkly-icon.rc" "${CMAKE_CURRENT_BINARY_DIR}/youngkly.ico"
      WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
    set(youngkly_icon "${CMAKE_CURRENT_BINARY_DIR}/youngkly-icon.o")
    set_source_files_properties("${youngkly_icon}" PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
  else()
    set(youngkly_icon "${CMAKE_CURRENT_BINARY_DIR}/youngkly-icon.rc")
  endif()
  qt_add_resources(youngkly_resources resources/youngkly.qrc)
  add_executable(youngkly_app WIN32 src/app_main.cpp src/FutureEditor.cpp ${youngkly_icon} ${youngkly_resources})
  target_link_libraries(youngkly_app PRIVATE youngkly_core Qt6::Widgets)
endif()
